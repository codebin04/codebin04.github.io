## 2번

```cpp
#include <iostream>
using namespace std;

int main() {
	int count = 0;
	cout << "입력>>";
	while (true) {
		char ch;
		cin.get(ch);
		if (cin.eof())
			break;

		if (ch == '\n') {
			cout << count << "개 입력됨" << endl;
			count = 0;
			cout << "입력>>";
		}
		else if (ch >= '0' && ch <= '9')
				count++;

	}
}
```

## 7번

```cpp

#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

class Phone {
	string name, telNum, address;
public:
	Phone(string name = "", string telNum = "", string address = "") {
		this->name = name;
		this->telNum = telNum;
		this->address = address;
	}

	friend istream& operator >> (istream& ist, Phone& p);
	friend ostream& operator << (ostream& ost, Phone p);
};

istream& operator >> (istream& ist, Phone& p) {
	cout << "이름:";
	ist >> p.name;
	cout << "전화번호:";
	ist >> p.telNum;
	cout << "주소:";
	ist.ignore(1);
	getline(ist, p.address);

	return ist;
}

ostream& operator << (ostream& ost, Phone p) {
	ost << "( " << p.name << ", " << p.telNum << ", " << p.address << " )";
	return ost;
}

int main() {
	Phone queen, prince;
	cin >> queen >> prince;
	cout << queen << endl << prince << endl;
}
```

## 8번

```cpp
#include <iostream>
using namespace std;

class Exp {
	int base;
	int exp;
public:
	Exp(int base = 1, int exp = 1);
	int calcValue();

	friend istream& operator >> (istream& ist, Exp& e);
	friend ostream& operator << (ostream& ost, Exp e);
};

Exp::Exp(int base, int exp) {
	this->base = base;
	this->exp = exp;
}

int Exp::calcValue() {
	int calc = 1;
	for (int i = 0; i < exp; i++) {
		calc *= base;
	}
	return calc;
}

istream& operator >> (istream& ist, Exp& e) {
	cout << "베이스: ";
	ist >> e.base;
	cout << "지수: ";
	ist >> e.exp;

	return ist;
}

ostream& operator << (ostream& ost, Exp e) {
	ost << e.base << "의 " << e.exp << "승:" << e.calcValue();
	return ost;
}

int main() {
	Exp a, b;
	cin >> a >> b;
	cout << a << endl << b << endl;
}
```
